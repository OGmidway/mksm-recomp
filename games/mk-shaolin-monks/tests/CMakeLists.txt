cmake_minimum_required(VERSION 3.21)
project(MkIopChecks LANGUAGES CXX)
get_filename_component(RECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PS2X_IOP_ENABLE_PLUGINS OFF CACHE BOOL "IOP plugin support" FORCE)
add_subdirectory("${RECOMP_ROOT}/ps2xIOP" "${CMAKE_CURRENT_BINARY_DIR}/ps2xIOP")
add_executable(mk_iop_checks main.cpp "${RECOMP_ROOT}/ps2xTest/src/ps2_iop_tests.cpp")
target_compile_features(mk_iop_checks PRIVATE cxx_std_20)
target_include_directories(mk_iop_checks PRIVATE "${RECOMP_ROOT}/ps2xTest/include")
option(MKSM_RETAIL_FIXTURES "Enable tests requiring locally generated retail bank/ADPCM fixtures" OFF)
if(MKSM_RETAIL_FIXTURES)
    target_compile_definitions(mk_iop_checks PRIVATE MKSM_RETAIL_FIXTURES=1)
endif()
target_link_libraries(mk_iop_checks PRIVATE ps2_iop)
